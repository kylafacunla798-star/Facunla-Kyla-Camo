import java.util.Scanner;
import java.util.Random;

public class WebsiteVisitorChallenge {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        Random random = new Random();

        int[] visitors = new int[10];
        int score = 0;
        int total = 0;
        int maxVisitors = 0;
        int maxIndex = 0;
        int above500 = 0;

        System.out.println("WEBSITE VISITOR CHALLENGE");
        System.out.println("1. More than 500 visitors");
        System.out.println("2. 500 visitors and below");

        for (int i = 0; i < visitors.length; i++) {
            visitors[i] = random.nextInt(1000) + 1;
        }

        for (int i = 0; i < visitors.length; i++) {
            int guess;

            System.out.println("\nDAY " + (i + 1));

            while (true) {
                System.out.print("Your guess (1 or 2): ");

                if (input.hasNextInt()) {
                    guess = input.nextInt();

                    if (guess == 1 || guess == 2) {
                        break;
                    }
                } else {
                    input.next();
                }

                System.out.println("Invalid input. Try again.");
            }

            System.out.println("Actual visitors: " + visitors[i]);

            if ((visitors[i] > 500 && guess == 1) ||
                (visitors[i] <= 500 && guess == 2)) {

                System.out.println("Correct!");
                score++;

            } else {
                System.out.println("Wrong guess!");
            }

            total += visitors[i];

            if (i == 0 || visitors[i] > maxVisitors) {
                maxVisitors = visitors[i];
                maxIndex = i;
            }

            if (visitors[i] > 500) {
                above500++;
            }
        }

        double average = total / 10.0;

        System.out.println("\nGAME RESULTS");
        System.out.println("Final Score: " + score + "/10");
        System.out.println("Total Visitors: " + total);
        System.out.println("Average Visitors: " + average);
        System.out.println("Busiest Day: Day " + (maxIndex + 1));
        System.out.println("Highest Visitors: " + maxVisitors);
        System.out.println("Days Above 500: " + above500);

        input.close();
    }
}
